balance = 2000

def deposit(amount):
    global balance
    if amount > 500:
        balance += amount
        print("Amount deposited successfully")

def withdraw(amount):
    global balance
    if amount <= 0:
        print("Invalid amount")
    elif amount <= balance:
        balance -= amount
        print("Amount withdrawn successfully")
    else:
        print("Insufficient balance")
